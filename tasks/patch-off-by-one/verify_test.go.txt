package main

import (
	"reflect"
	"testing"
)

func TestWindow(t *testing.T) {
	cases := []struct {
		xs   []int
		n    int
		want []int
	}{
		{[]int{1, 2, 3, 4, 5}, 2, []int{4, 5}},
		{[]int{1, 2, 3}, 3, []int{1, 2, 3}},
		{[]int{1, 2, 3}, 9, []int{1, 2, 3}},
		{[]int{1, 2, 3}, 1, []int{3}},
		{[]int{1, 2, 3}, 0, []int{}},
		{[]int{1, 2, 3}, -1, []int{}},
		{[]int{}, 2, []int{}},
	}
	for _, c := range cases {
		got := Window(c.xs, c.n)
		// A nil slice and an empty one are both "an empty slice" — len 0, ranges and
		// appends identically — and the doc comment does not distinguish them.
		// reflect.DeepEqual does, so comparing with it alone would fail an answer that
		// is correct about the off-by-one this task is actually asking about.
		if len(got) == 0 && len(c.want) == 0 {
			continue
		}
		if !reflect.DeepEqual(got, c.want) {
			t.Errorf("Window(%v, %d) = %v, want %v", c.xs, c.n, got, c.want)
		}
	}
}
